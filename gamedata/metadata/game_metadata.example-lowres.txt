# ===========================================================
# Low Resolution Game at 64x64 pixels
# ===========================================================
# Perfect for LOWREZ Jam: https://itch.io/jam/lowrezjam-2026
# Sample Apps:
# - ./samples/99_genre_lowrez/resolution_64x64
# ===========================================================

# configuration that's needed for any release
devid=your_itchio_username
devtitle=Your Itch.io Display Name
gameid=your_itchio_game_id
gametitle=Your Game Title
version=0.1
icon=metadata/icon.png

# aspect_mode=1 means 1:1 aspect ratio
aspect_mode=1

# aspect_size=64 means 64 pixels width and height
aspect_size=64

# a scale quality of 3 means "pixel art scaling"
scale_quality=3

# Pro version of DragonRuby will give you extremely sharp sprites
highdpi=true
